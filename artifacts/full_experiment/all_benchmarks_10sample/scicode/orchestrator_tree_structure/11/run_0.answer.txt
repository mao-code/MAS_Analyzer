 # Background: In quantum mechanics, a standard basis vector |j> in a d-dimensional Hilbert space is represented as a column vector with a 1 at the j-th index (0-indexed) and 0 elsewhere. For composite systems, the state space is the tensor product of individual Hilbert spaces. The tensor product of vectors |v1> and |v2> is computed using the Kronecker product, which results in a vector of dimension d1*d2. def ket(dim, j): '''Input: dim: int or list, dimension of the ket j: int or list, the i-th basis vector index Output: out: dim dimensional array of float, the matrix representation of the ket ''' if isinstance(dim, int): out = np.zeros(dim) out[j] = 1.0 return out else: # Handle list of dimensions and list of indices kets = [] for d_i, j_i in zip(dim, j): vec = np.zeros(d_i) vec[j_i] = 1.0 kets.append(vec) # Compute tensor product iteratively out = kets[0] for i in range(1, len(kets)): out = np.kron(out, kets[i]) return out
def multi_rail_encoding_state(rails): # Background: In m-rail encoding, a logical state is represented by the presence of a photon in one of m rails. A bipartite maximally entangled state in this context is defined as |Phi> = (1/sqrt(rails)) * sum_{i=0}^{rails-1} (|i>_A ⊗ |i>_B), where |i> is the basis vector with a 1 at index i. The density matrix is given by |Phi><Phi|. dim = rails state_vec = np.zeros(dim * dim) # Construct the state vector |Phi> = (1/sqrt(m)) * sum(|i>|i>) for i in range(dim): ket_i = np.zeros(dim) ket_i[i] = 1.0 # Tensor product of |i>_A and |i>_B basis_pair = np.kron(ket_i, ket_i) state_vec += basis_pair state_vec /= np.sqrt(dim) # Density matrix rho = |Phi><Phi| state = np.outer(state_vec, state_vec) return state
 # Background: The tensor product (or Kronecker product) of multiple matrices A, B, C, ... is an associative operation. For a sequence of matrices M_1, M_2, ..., M_n, the tensor product is defined as M_1 ⊗ M_2 ⊗ ... ⊗ M_n. In numerical computing, this can be computed iteratively by taking the Kronecker product of the current result with the next matrix in the sequence: R_1 = M_1, R_k = R_{k-1} ⊗ M_k. def tensor(*args): '''Takes the tensor product of an arbitrary number of matrices/vectors. Input: args: any number of nd arrays of floats, corresponding to input matrices Output: M: the tensor product (kronecker product) of input matrices, 2d array of floats ''' if not args: return None M = args[0] for i in range(1, len(args)): M = np.kron(M, args[i]) return M
 # Background: A quantum channel represented by Kraus operators {K_i} transforms a density matrix rho according to the map E(rho) = sum_i K_i * rho * K_i^dagger. When a channel acts on specific subsystems of a larger composite system, the Kraus operators must be extended to the full Hilbert space. If the channel acts on a subset of systems 'sys' with dimensions 'dim', the effective Kraus operator K_eff is the tensor product of the channel operator K acting on the specified subsystems and identity operators acting on the remaining subsystems. The final state is the sum over all Kraus operators of K_eff * rho * K_eff^dagger. def apply_channel(K, rho, sys=None, dim=None): '''Applies the channel with Kraus operators in K to the state rho on systems specified by the list sys. The dimensions of the subsystems of rho are given by dim. ''' if sys is None or dim is None: # Apply to the entire system return sum(k @ rho @ k.conj().T for k in K) # Identify total number of systems num_systems = len(dim) # Construct the full-space Kraus operators full_K = [] for k in K: # Create a list of operators for each subsystem ops = [np.eye(d) for d in dim] # Replace the identity with the channel operator for the specified subsystems # Assuming K acts on the joint space of the subsystems in 'sys' # We place the operator at the first index of the sys list for simplicity # or assume K is already defined for the joint space of sys ops[sys[0]] = k # Compute the tensor product of all operators k_full = ops[0] for i in range(1, num_systems): k_full = np.kron(k_full, ops[i]) full_K.append(k_full) # Apply the Kraus map return sum(k @ rho @ k.conj().T for k in full_K)
def generalized_amplitude_damping_channel(gamma, N):
    """Generates the generalized amplitude damping channel. Inputs: gamma: float, damping parameter N: float, thermal parameter Output: kraus: list of Kraus operators as 2x2 arrays of floats, [A1, A2, A3, A4] """
 # Background: The generalized amplitude damping channel (GADC) models energy dissipation into a thermal reservoir. For a single qubit, the Kraus operators are: E0 = sqrt(p) * [[1, 0], [0, sqrt(1-gamma)]], E1 = sqrt(p) * [[0, sqrt(gamma)], [0, 0]], E2 = sqrt(1-p) * [[sqrt(1-gamma), 0], [0, 1]], E3 = sqrt(1-p) * [[0, 0], [sqrt(gamma), 0]], where p = 1/(N+1) is related to the thermal photon number N. For an m-rail encoded state, the system consists of 2*m qubits (m for Alice, m for Bob). The channel acts on each rail independently. The total Kraus operators for the composite system are the tensor products of the individual Kraus operators for each of the 2*m rails. def output_state(rails, gamma_1, N_1, gamma_2, N_2): # 1. Construct the initial bipartite maximally entangled state |Phi> = (1/sqrt(rails)) * sum_{i=0}^{rails-1} (|i>_A |i>_B) # In m-rail encoding, |i> is a basis vector of length m with 1 at index i. dim = rails state_vec = np.zeros(dim * dim) for i in range(dim): ket_i = np.zeros(dim) ket_i[i] = 1.0 state_vec += np.kron(ket_i, ket_i) state_vec /= np.sqrt(dim) rho = np.outer(state_vec, state_vec) # 2. Define Kraus operators for GADC def get_kraus(gamma, N): p = 1.0 / (N + 1.0) E0 = np.sqrt(p) * np.array([[1, 0], [0, np.sqrt(1 - gamma)]]) E1 = np.sqrt(p) * np.array([[0, np.sqrt(gamma)], [0, 0]]) E2 = np.sqrt(1 - p) * np.array([[np.sqrt(1 - gamma), 0], [0, 1]]) E3 = np.sqrt(1 - p) * np.array([[0, 0], [np.sqrt(gamma), 0]]) return [E0, E1, E2, E3] K1 = get_kraus(gamma_1, N_1) K2 = get_kraus(gamma_2, N_2) # 3. Apply channels to each rail # Total rails = 2 * rails. First 'rails' go through channel 1, second 'rails' through channel 2. # We apply the channel to each subsystem individually. current_rho = rho for i in range(rails): current_rho = apply_channel(K1, current_rho, sys=[i], dim=[2]*(2*rails)) for i in range(rails, 2*rails): current_rho = apply_channel(K2, current_rho, sys=[i], dim=[2]*(2*rails)) return current_rho

    """Returns the measurement projector Input: rails: int, number of rails Output: global_proj: ( 2**(2*rails), 2**(2*rails) ) dimensional array of floats """
 # Background: Permuting subsystems of a multipartite quantum state represented by a density matrix X involves rearranging the tensor product structure of the Hilbert space. Given a state X in a Hilbert space H = H_0 ⊗ H_1 ⊗ ... ⊗ H_{n-1} with dimensions dim = [d_0, d_1, ..., d_{n-1}], a permutation 'perm' reorders the subsystems such that the new state Y corresponds to the Hilbert space H_{perm[0]} ⊗ H_{perm[1]} ⊗ ... ⊗ H_{perm[n-1]}. To achieve this, the density matrix X is reshaped into a tensor of rank 2n with shape (d_0, ..., d_{n-1}, d_0, ..., d_{n-1}). The indices are then permuted according to the specified order for both the row and column spaces, and the resulting tensor is reshaped back into a 2D matrix. def syspermute(X, perm, dim): '''Permutes order of subsystems in the multipartite operator X. Inputs: X: 2d array of floats with equal dimensions, the density matrix of the state perm: list of int containing the desired order dim: list of int containing the dimensions of all subsystems. Output: Y: 2d array of floats with equal dimensions, the density matrix of the permuted state ''' n = len(dim) # Reshape into a 2n-rank tensor: (d_0, ..., d_{n-1}, d_0, ..., d_{n-1}) tensor_shape = tuple(dim) + tuple(dim) X_tensor = X.reshape(tensor_shape) # The permutation applies to both row indices (0 to n-1) and column indices (n to 2n-1) # New order: [perm[0], ..., perm[n-1], perm[0]+n, ..., perm[n-1]+n] full_perm = list(perm) + [p + n for p in perm] # Transpose the tensor according to the permutation Y_tensor = np.transpose(X_tensor, full_perm) # Reshape back to 2D matrix new_dim = [dim[i] for i in perm] total_dim = np.prod(new_dim) Y = Y_tensor.reshape((total_dim, total_dim)) return Y
 # Background: The partial trace of a density matrix X over a subsystem B is defined as Tr_B(X) = sum_i (I_A ⊗ <i|_B) X (I_A ⊗ |i>_B), where {|i>_B} is an orthonormal basis for the Hilbert space of subsystem B. Computationally, this involves permuting the subsystems such that the systems to be traced out are at the end of the tensor product structure. After permutation, the density matrix is reshaped into a 4-index tensor (A, B, A, B), where A represents the kept systems and B represents the traced systems. The partial trace is then computed by summing over the diagonal elements of the B indices: Tr_B(X) = sum_i (A, i, A, i). def partial_trace(X, sys, dim): '''Inputs: X: 2d array of floats with equal dimensions, the density matrix of the state sys: list of int containing systems over which to take the partial trace (i.e., the systems to discard). dim: list of int containing dimensions of all subsystems. Output: 2d array of floats with equal dimensions, density matrix after partial trace. ''' n = len(dim) # Identify systems to keep keep = [i for i in range(n) if i not in sys] # Permute systems so that 'keep' systems come first, then 'sys' systems perm = keep + sys X_perm = syspermute(X, perm, dim) # Dimensions of kept and traced systems dim_keep = [dim[i] for i in keep] dim_sys = [dim[i] for i in sys] size_keep = np.prod(dim_keep) size_sys = np.prod(dim_sys) # Reshape into (size_keep, size_sys, size_keep, size_sys) tensor = X_perm.reshape((size_keep, size_sys, size_keep, size_sys)) # Partial trace: sum over the diagonal of the traced systems (the second and fourth indices) # Result is sum_i tensor[:, i, :, i] out = np.zeros((size_keep, size_keep), dtype=complex) for i in range(size_sys): out += tensor[:, i, :, i] return out
 # Background: The von Neumann entropy of a quantum state represented by a density matrix rho is defined as S(rho) = -Tr(rho * log2(rho)). This is calculated by finding the eigenvalues {lambda_i} of the density matrix rho and computing the sum S(rho) = -sum_{i} lambda_i * log2(lambda_i), where terms with lambda_i = 0 are treated as 0 since lim_{x->0} x*log(x) = 0. For a density matrix, which is Hermitian and positive semi-definite, scipy.linalg.eigvalsh is used to compute the eigenvalues efficiently. def entropy(rho): '''Inputs: rho: 2d array of floats with equal dimensions, the density matrix of the state Output: en: quantum (von Neumann) entropy of the state rho, float ''' # Calculate eigenvalues of the Hermitian density matrix eigenvalues = scipy.linalg.eigvalsh(rho) # Filter out eigenvalues that are effectively zero to avoid log(0) errors # We use a small tolerance for numerical stability nonzero_eigenvalues = eigenvalues[eigenvalues > 1e-15] # Compute entropy: -sum(lambda * log2(lambda)) en = -np.sum(nonzero_eigenvalues * np.log2(nonzero_eigenvalues)) return float(en)
 # Background: The coherent information of a bipartite quantum state rho_AB is defined as I(A>B) = S(rho_B) - S(rho_AB), where S(rho) = -Tr(rho log2 rho) is the von Neumann entropy. It quantifies the amount of quantum information that can be transmitted from system A to system B through a quantum channel. Here, S(rho_AB) is the entropy of the joint state, and S(rho_B) is the entropy of the reduced density matrix of system B, obtained by tracing out system A. def coherent_inf_state(rho_AB, dimA, dimB): '''Inputs: rho_AB: 2d array of floats with equal dimensions, the state we evaluate coherent information dimA: int, dimension of system A dimB: int, dimension of system B Output co_inf: float, the coherent information of the state rho_AB ''' # Calculate the entropy of the joint state S(rho_AB) s_ab = entropy(rho_AB) # Calculate the reduced density matrix rho_B = Tr_A(rho_AB) # The system A is at index 0, system B is at index 1 rho_b = partial_trace(rho_AB, [0], [dimA, dimB]) # Calculate the entropy of the reduced state S(rho_B) s_b = entropy(rho_b) # Coherent information I(A>B) = S(rho_B) - S(rho_AB) co_inf = s_b - s_ab return float(co_inf)

def rate(rails, gamma_1, N_1, gamma_2, N_2): ''' Background: The hashing protocol allows for the distillation of entanglement from noisy states. The achievable rate of entanglement distillation is given by the coherent information I(A>B) = S(rho_B) - S(rho_AB). In this protocol, we first apply the generalized amplitude damping channels to the m-rail encoded state, then perform a post-selection measurement to project the state into the one-particle sector (where each party has exactly one photon across their m rails). The resulting state is normalized, and its coherent information is calculated to determine the entanglement rate. ''' # 1. Get the state after passing through the channels rho_noisy = output_state(rails, gamma_1, N_1, gamma_2, N_2) # 2. Construct the projector for the one-particle sector # The one-particle sector for m rails is the subspace spanned by basis vectors with exactly one 1. # There are 'rails' such vectors for each party. Total dimension is (2*rails) qubits. # The projector P = P_A ⊗ P_B, where P_A = sum_{i=0}^{rails-1} |i><i| def get_one_particle_projector(m): proj = np.zeros((m, m)) for i in range(m): ket = np.zeros(m) ket[i] = 1.0 proj += np.outer(ket, ket) return proj P_A = get_one_particle_projector(rails) P_B = get_one_particle_projector(rails) P = np.kron(P_A, P_B) # 3. Apply the projector: rho_post = P * rho * P^† rho_post = P @ rho_noisy @ P.conj().T # 4. Normalize the state prob = np.trace(rho_post) if prob < 1e-15: return 0.0 rho_post /= prob # 5. Calculate coherent information # System A is the first 'rails' qubits, System B is the second 'rails' qubits. # Note: The input state to coherent_inf_state expects dimensions of subsystems. # Here, each party has 'rails' rails, but the state is in 2*rails qubits space. # The coherent_inf_state function uses partial_trace which expects subsystem dimensions. return coherent_inf_state(rho_post, rails, rails)
