import numpy as np


 # Background: The Lanczos iteration is an algorithm used to compute the eigenvalues and eigenvectors of a large, sparse symmetric matrix. It constructs an orthonormal basis for the Krylov subspace K_m(A, b) = span{b, Ab, A^2b, ..., A^{m-1}b}. The process uses a three-term recurrence relation to generate the basis vectors (Lanczos vectors), ensuring orthogonality through Gram-Schmidt orthogonalization. Given a symmetric matrix A and an initial vector b, the algorithm produces a sequence of orthonormal vectors q_1, q_2, ..., q_m such that the matrix Q = [q_1, ..., q_m] satisfies Q^T A Q = T, where T is a tridiagonal matrix. import numpy as np def lanczos(A, b, m): '''Inputs: A : Matrix, 2d array of arbitrary size M * M b : Vector, 1d array of arbitrary size M * 1 m : integer, m < M Outputs: Q : Matrix, 2d array of size M*(m+1) ''' M = A.shape[0] Q = np.zeros((M, m + 1)) # Normalize the initial vector q = b / np.linalg.norm(b) Q[:, 0] = q beta = 0 q_prev = np.zeros(M) for j in range(m): v = A @ Q[:, j] if j > 0: v = v - beta * q_prev alpha = np.dot(Q[:, j], v) v = v - alpha * Q[:, j] # Re-orthogonalization (optional but recommended for numerical stability) for i in range(j + 1): v = v - np.dot(Q[:, i], v) * Q[:, i] beta = np.linalg.norm(v) if beta < 1e-12: break q_prev = Q[:, j] Q[:, j + 1] = v / beta return Q
