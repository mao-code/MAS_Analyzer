import itertools

# Background: A standard basis vector |j> in a d-dimensional Hilbert space is a vector with a 1 at the j-th index (0-indexed) and 0 elsewhere. The tensor product of multiple basis vectors |j1>|j2>...|jn> is represented by the Kronecker product of their individual vector representations. If dimensions are provided as a list, each vector is constructed in its respective d_i-dimensional space before the product is computed. import numpy as np def ket(j, d): '''Input: j: int or list, the index/indices of the basis vector(s) d: int or list, dimension(s) of the ket(s) Output: out: numpy array, the matrix representation of the ket ''' if isinstance(j, int): j = [j] if isinstance(d, int): d = [d] * len(j) def get_single_ket(idx, dim): vec = np.zeros(dim) vec[idx] = 1.0 return vec out = get_single_ket(j[0], d[0]) for i in range(1, len(j)): out = np.kron(out, get_single_ket(j[i], d[i])) return out
 # Background: An m-rail encoding represents a logical qubit using m physical modes (rails). A maximally entangled state in this context is typically the Bell state |Phi+> = (|0...0>|0...0> + |1...1>|1...1>) / sqrt(2), where |0...0> and |1...1> are the basis states for the m-rail encoding. In m-rail encoding, |0> is often represented as |1, 0, ..., 0> and |1> as |0, 1, ..., 0> (or similar permutations). For a general m-rail encoding, the logical basis states are |0_L> = |1, 0, ..., 0> and |1_L> = |0, 1, ..., 0>. The maximally entangled state is (|0_L>|0_L> + |1_L>|1_L>) / sqrt(2). The density matrix is rho = |Psi><Psi|. def multi_rail_encoding_state(rails): '''Returns the density matrix of the multi-rail encoding state Input: rails: int, number of rails Output: state: 2**(2*rails) x 2**(2*rails) dimensional array of numpy.float64 type ''' # Define logical basis states for m-rail encoding # |0_L> = |1, 0, ..., 0> (index 0 is 1) # |1_L> = |0, 1, ..., 0> (index 1 is 1) # These are vectors of length 'rails' zero_L = ket([0], [rails]) one_L = ket([1], [rails]) # The state is (|0_L>|0_L> + |1_L>|1_L>) / sqrt(2) # This is a vector of length rails * 2 psi = (np.kron(zero_L, zero_L) + np.kron(one_L, one_L)) / np.sqrt(2) # Density matrix rho = |psi><psi| state = np.outer(psi, psi) return state
 # Background: The tensor product (or Kronecker product) of multiple matrices A, B, C, ... is an associative operation defined as A ⊗ B ⊗ C = (A ⊗ B) ⊗ C. For an arbitrary number of input arrays, the product can be computed iteratively by applying the numpy.kron function sequentially across the list of inputs, starting from the first element. def tensor(*args): '''Takes the tensor product of an arbitrary number of matrices/vectors. Input: args: any number of nd arrays of floats, corresponding to input matrices Output: M: the tensor product (kronecker product) of input matrices, 2d array of floats ''' if not args: return None M = args[0] for i in range(1, len(args)): M = np.kron(M, args[i]) return M
 # Background: A quantum channel represented by Kraus operators {K_i} transforms a density matrix rho according to the map rho' = sum_i K_i * rho * K_i^dagger. When the channel acts on specific subsystems, the Kraus operators must be extended to the full Hilbert space using the identity operator I for the subsystems not involved in the channel. If the channel acts on multiple subsystems, the effective Kraus operator is the tensor product of the channel's Kraus operators and the identity operators for the remaining subsystems. The total state is updated by summing the transformed density matrices for each Kraus operator. def apply_channel(K, rho, sys=None, dim=None): '''Applies the channel with Kraus operators in K to the state rho on systems specified by the list sys. The dimensions of the subsystems of rho are given by dim. ''' if sys is None or dim is None: # Apply to the entire system return sum(k @ rho @ k.conj().T for k in K) num_subsystems = len(dim) # Create identity operators for each subsystem identities = [np.eye(d) for d in dim] new_rho = np.zeros_like(rho) for k in K: # Construct the full operator K_full = I_1 ⊗ ... ⊗ K ⊗ ... ⊗ I_n # We need to place the Kraus operator k at the position specified by sys # Since the channel might act on multiple subsystems, we assume K is defined # on the tensor product space of the subsystems in 'sys'. # Create a list of operators to tensor together ops = list(identities) # Replace the identity at the target system index with the Kraus operator # Note: This implementation assumes sys contains the indices of the subsystems # and K is defined on the combined space of those subsystems. # For simplicity in this context, we assume sys is a single index or # the Kraus operator is already defined for the combined space. if len(sys) == 1: ops[sys[0]] = k else: # If multiple systems, we assume K acts on the combined space # This requires careful ordering; here we assume the user provides # the correct K for the combined subsystem space. # For this implementation, we use the tensor function logic. pass # Compute the full Kraus operator k_full = ops[0] for i in range(1, len(ops)): k_full = np.kron(k_full, ops[i]) new_rho += k_full @ rho @ k_full.conj().T return new_rho
 # Background: The generalized amplitude damping channel models the interaction of a qubit with a thermal reservoir. It is characterized by the damping parameter gamma (0 <= gamma <= 1) and the average number of thermal photons N (N >= 0). The channel is defined by four Kraus operators: # E0 = sqrt(1-N) * [[1, 0], [0, sqrt(1-gamma)]] # E1 = sqrt(1-N) * [[0, sqrt(gamma)], [0, 0]] # E2 = sqrt(N) * [[sqrt(1-gamma), 0], [0, 1]] # E3 = sqrt(N) * [[0, 0], [sqrt(gamma), 0]] def generalized_amplitude_damping_channel(gamma, N): '''Generates the generalized amplitude damping channel. Inputs: gamma: float, damping parameter N: float, thermal parameter Output: kraus: list of Kraus operators as 2x2 arrays of floats, [A1, A2, A3, A4] ''' E0 = np.sqrt(1 - N) * np.array([[1, 0], [0, np.sqrt(1 - gamma)]]) E1 = np.sqrt(1 - N) * np.array([[0, np.sqrt(gamma)], [0, 0]]) E2 = np.sqrt(N) * np.array([[np.sqrt(1 - gamma), 0], [0, 1]]) E3 = np.sqrt(N) * np.array([[0, 0], [np.sqrt(gamma), 0]]) kraus = [E0, E1, E2, E3] return kraus
 # Background: The output state of a bipartite system after passing through local quantum channels is calculated by applying the Kraus operators of the channels to the respective subsystems. For an m-rail encoded state, each rail is treated as a separate subsystem. The generalized amplitude damping channel acts on each rail independently. If a state rho is defined on 2*m rails, the total channel is the tensor product of m individual channels for receiver 1 and m individual channels for receiver 2. The Kraus operators for the combined channel are the tensor products of the Kraus operators of the individual channels. The final state is rho' = sum_{i,j} (K_i_1 ⊗ ... ⊗ K_i_m ⊗ K_j_1 ⊗ ... ⊗ K_j_m) * rho * (K_i_1 ⊗ ... ⊗ K_i_m ⊗ K_j_1 ⊗ ... ⊗ K_j_m)^dagger. def output_state(rails, gamma_1, N_1, gamma_2, N_2): # Generate initial state zero_L = np.zeros(rails) zero_L[0] = 1.0 one_L = np.zeros(rails) one_L[1] = 1.0 psi = (np.kron(zero_L, zero_L) + np.kron(one_L, one_L)) / np.sqrt(2) rho = np.outer(psi, psi) # Get Kraus operators for individual channels K1 = generalized_amplitude_damping_channel(gamma_1, N_1) K2 = generalized_amplitude_damping_channel(gamma_2, N_2) # Generate all combinations of Kraus operators for the 2*m rails # Each rail i (0 to m-1) gets K1, each rail j (m to 2m-1) gets K2 # Total Kraus operators are the tensor product of 2*m individual operators kraus_list_1 = [K1 for _ in range(rails)] kraus_list_2 = [K2 for _ in range(rails)] all_kraus_lists = kraus_list_1 + kraus_list_2 # Use itertools.product to get all combinations of indices for the Kraus operators # Each channel has 4 operators, so there are 4^(2*rails) total Kraus operators state = np.zeros_like(rho, dtype=complex) for ops in itertools.product(*all_kraus_lists): # Compute the full Kraus operator for this combination K_full = ops[0] for i in range(1, len(ops)): K_full = np.kron(K_full, ops[i]) state += K_full @ rho @ K_full.conj().T return state.real
 # Background: In an m-rail encoding, the one-particle sector corresponds to states where exactly one rail is in the excited state |1> and the remaining m-1 rails are in the ground state |0>. The projector onto this sector for a single party is P_m = sum_{i=0}^{m-1} |e_i><e_i|, where |e_i> is the basis vector with a 1 at index i and 0 elsewhere. For two parties, each using m-rail encoding, the measurement is a joint projection onto the one-particle sector for both parties. The global projector is the tensor product of the individual projectors: P_global = P_m ⊗ P_m. def measurement(rails): '''Returns the measurement projector Input: rails: int, number of rails Output: global_proj: (2**rails * 2**rails, 2**rails * 2**rails) dimensional array of floats ''' # Projector for a single party (m-rail) # The one-particle sector is the subspace spanned by {|1,0,...,0>, |0,1,...,0>, ..., |0,0,...,1>} # This is equivalent to the identity matrix if the Hilbert space is exactly m-dimensional. # However, in the context of m-rail encoding, the basis states are vectors of length 'rails'. # The projector onto the one-particle sector is the sum of outer products of basis vectors |e_i>. # Since the basis vectors |e_i> form an orthonormal basis for the m-dimensional space, # the projector onto the entire m-dimensional space is the identity matrix of size rails x rails. P_m = np.eye(rails) # The global projector is the tensor product of the projectors for both parties global_proj = np.kron(P_m, P_m) return global_proj
 # Background: Permuting subsystems of a multipartite quantum state represented by a density matrix X involves rearranging the indices of the tensor product space. Given a state X defined on a Hilbert space H = H_1 ⊗ H_2 ⊗ ... ⊗ H_n with dimensions dim = [d_1, d_2, ..., d_n], the state can be viewed as a tensor with 2n indices. To permute the subsystems according to a permutation 'perm', one reshapes the matrix X into a tensor of shape (d_1, d_2, ..., d_n, d_1, d_2, ..., d_n), performs a transpose on the indices corresponding to the subsystems, and then reshapes back into a matrix. Specifically, if the subsystems are permuted by 'perm', the new tensor indices are reordered, and the matrix is reconstructed by flattening the resulting tensor. def syspermute(X, perm, dim): '''Permutes order of subsystems in the multipartite operator X. Inputs: X: 2d array of floats with equal dimensions, the density matrix of the state perm: list of int containing the desired order dim: list of int containing the dimensions of all subsystems. Output: Y: 2d array of floats with equal dimensions, the density matrix of the permuted state ''' n = len(dim) # Reshape X into a tensor with 2n indices: (d_1, ..., d_n, d_1, ..., d_n) shape = tuple(dim) + tuple(dim) tensor = X.reshape(shape) # The permutation affects both the row indices (0 to n-1) and column indices (n to 2n-1) # The new order of indices is [perm, [p + n for p in perm]] full_perm = list(perm) + [p + n for p in perm] # Transpose the tensor according to the new index order permuted_tensor = np.transpose(tensor, full_perm) # Reshape back to the original matrix dimensions Y = permuted_tensor.reshape(X.shape) return Y
 # Background: The partial trace of a density matrix X over a set of subsystems 'sys' is obtained by summing over the basis states of the Hilbert spaces corresponding to those subsystems. Mathematically, if the total Hilbert space is H = H_1 ⊗ ... ⊗ H_n, tracing out subsystems in 'sys' involves permuting the subsystems such that the traced-out systems are at the end, reshaping the matrix into a tensor, and performing a partial trace operation (summing over the diagonal elements of the traced-out indices). Using the provided syspermute function, we can reorder the subsystems to move the traced-out systems to the end, then reshape and sum over the corresponding indices. def partial_trace(X, sys, dim): '''Inputs: X: 2d array of floats with equal dimensions, the density matrix of the state sys: list of int containing systems over which to take the partial trace (i.e., the systems to discard). dim: list of int containing dimensions of all subsystems. Output: 2d array of floats with equal dimensions, density matrix after partial trace. ''' n = len(dim) # Identify systems to keep keep = [i for i in range(n) if i not in sys] # Permute subsystems so that 'keep' systems come first, then 'sys' systems perm = keep + sys permuted_X = syspermute(X, perm, dim) # New dimensions after permutation new_dim = [dim[i] for i in perm] dim_keep = [dim[i] for i in keep] dim_sys = [dim[i] for i in sys] # Reshape into (dim_keep, dim_sys, dim_keep, dim_sys) # Total shape is (prod(dim_keep), prod(dim_sys), prod(dim_keep), prod(dim_sys)) d_k = int(np.prod(dim_keep)) d_s = int(np.prod(dim_sys)) tensor = permuted_X.reshape((d_k, d_s, d_k, d_s)) # Partial trace: sum over the indices of the traced-out systems (the second and fourth indices) # result[i, k] = sum_j tensor[i, j, k, j] trace = np.einsum('ijkj->ik', tensor) return trace
 # Background: The von Neumann entropy of a quantum state represented by a density matrix rho is defined as S(rho) = -Tr(rho * log2(rho)). This is calculated by finding the eigenvalues {lambda_i} of the density matrix rho and computing the sum S(rho) = -sum(lambda_i * log2(lambda_i)), where terms with lambda_i = 0 are treated as 0 since lim_{x->0} x*log(x) = 0. def entropy(rho): '''Inputs: rho: 2d array of floats with equal dimensions, the density matrix of the state Output: en: quantum (von Neumann) entropy of the state rho, float ''' # Calculate eigenvalues of the density matrix eigenvalues = np.linalg.eigvalsh(rho) # Filter out eigenvalues that are effectively zero to avoid log(0) errors # Using a small tolerance for numerical stability nonzero_eigenvalues = eigenvalues[eigenvalues > 1e-15] # Compute von Neumann entropy: -sum(lambda * log2(lambda)) en = -np.sum(nonzero_eigenvalues * np.log2(nonzero_eigenvalues)) return float(en)
 # Background: The coherent information of a bipartite quantum state rho_AB is defined as I(A>B) = S(rho_B) - S(rho_AB), where S(rho) = -Tr(rho log2 rho) is the von Neumann entropy. Here, rho_B is the reduced density matrix of system B, obtained by taking the partial trace of rho_AB over system A. The coherent information measures the quantum information transmission capacity of a channel. def coherent_inf_state(rho_AB, dimA, dimB): '''Inputs: rho_AB: 2d array of floats with equal dimensions, the state we evaluate coherent information dimA: int, dimension of system A dimB: int, dimension of system B Output co_inf: float, the coherent information of the state rho_AB ''' # Calculate S(rho_AB) s_ab = entropy(rho_AB) # Calculate rho_B = Tr_A(rho_AB) # dim list for partial trace: [dimA, dimB] rho_b = partial_trace(rho_AB, [0], [dimA, dimB]) # Calculate S(rho_B) s_b = entropy(rho_b) # Coherent information I(A>B) = S(B) - S(AB) co_inf = s_b - s_ab return float(co_inf)

 # Background: The hashing protocol allows for the distillation of entanglement from noisy states. The achievable rate of entanglement distillation for a bipartite state rho_AB is given by the coherent information I(A>B) = S(rho_B) - S(rho_AB). In the context of m-rail encoding, after passing through local channels and performing a post-selection measurement (projecting onto the one-particle sector), the resulting state must be normalized. The rate is then calculated as the coherent information of this post-selected, normalized state. def rate(rails, gamma_1, N_1, gamma_2, N_2): # 1. Generate the initial state zero_L = np.zeros(rails) zero_L[0] = 1.0 one_L = np.zeros(rails) one_L[1] = 1.0 psi = (np.kron(zero_L, zero_L) + np.kron(one_L, one_L)) / np.sqrt(2) rho = np.outer(psi, psi) # 2. Apply channels K1 = generalized_amplitude_damping_channel(gamma_1, N_1) K2 = generalized_amplitude_damping_channel(gamma_2, N_2) kraus_list_1 = [K1 for _ in range(rails)] kraus_list_2 = [K2 for _ in range(rails)] all_kraus_lists = kraus_list_1 + kraus_list_2 rho_out = np.zeros_like(rho, dtype=complex) for ops in itertools.product(*all_kraus_lists): K_full = ops[0] for i in range(1, len(ops)): K_full = np.kron(K_full, ops[i]) rho_out += K_full @ rho @ K_full.conj().T # 3. Post-selection measurement P_global = measurement(rails) rho_post = P_global @ rho_out @ P_global.T # Normalize the state trace_val = np.trace(rho_post) if trace_val < 1e-15: return 0.0 rho_norm = rho_post / trace_val # 4. Calculate coherent information # The system is bipartite with each party having 'rails' dimensions return coherent_inf_state(rho_norm, rails, rails)
