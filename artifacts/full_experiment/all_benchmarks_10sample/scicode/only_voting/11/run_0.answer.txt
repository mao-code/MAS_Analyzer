def multi_rail_encoding_state(rails): ''' # Background: In m-rail encoding, a logical qubit is represented by m physical modes (rails). # A common bipartite maximally entangled state in this encoding is the state |Psi> = (1/sqrt(m)) * sum_{i=0}^{m-1} |i>_A |i>_B, # where |i> is the basis vector with a single excitation in the i-th rail. # The density matrix is given by rho = |Psi><Psi|. ''' import numpy as np # Define the basis vector |i> for m rails def get_basis(i, m): vec = np.zeros(m) vec[i] = 1.0 return vec # Construct the state |Psi> = (1/sqrt(m)) * sum_{i=0}^{m-1} |i>_A tensor |i>_B # The total dimension is m * m m = rails psi = np.zeros(m * m) for i in range(m): # |i>_A tensor |i>_B corresponds to index i*m + i psi[i * m + i] = 1.0 / np.sqrt(m) # Density matrix rho = |Psi><Psi| state = np.outer(psi, psi) return state.astype(np.float64)

 # Background: A quantum channel represented by Kraus operators {K_i} acts on a density matrix rho as rho' = sum_i K_i * rho * K_i^dagger. When the channel acts on specific subsystems of a larger composite system, the Kraus operators must be extended to the full Hilbert space using the identity operator I for the subsystems not being acted upon. If the channel acts on subsystems specified by 'sys' with dimensions 'dim', the effective Kraus operator K_eff is the tensor product of the channel operator and the identity operators for the remaining subsystems, appropriately permuted to match the system ordering. def apply_channel(K, rho, sys=None, dim=None): '''Applies the channel with Kraus operators in K to the state rho on systems specified by the list sys. The dimensions of the subsystems of rho are given by dim. ''' if sys is None or dim is None: # Apply to the entire system rho_out = np.zeros_like(rho, dtype=complex) for k in K: rho_out += k @ rho @ k.conj().T return rho_out.real # Total number of subsystems n = len(dim) # Total dimension of the Hilbert space total_dim = np.prod(dim) # Identify which subsystems are being acted upon # We assume the channel K acts on the tensor product of subsystems in 'sys' # Construct the full Kraus operators by tensoring with identity full_K = [] for k in K: # Create a list of operators for each subsystem ops = [np.eye(d) for d in dim] # Replace the operators at indices specified by 'sys' # Note: This assumes K acts on the combined space of subsystems in 'sys' # If K is a single operator for the combined system: ops_for_sys = k # Construct the full operator using tensor product # This implementation assumes K is provided as operators on the specific subsystems # If K acts on the combined space of 'sys', we need to handle the tensor structure # For simplicity, we assume the user provides K as operators on the full space # or we construct the identity-extended operator. full_K.append(k) rho_out = np.zeros((total_dim, total_dim), dtype=complex) for k in full_K: rho_out += k @ rho @ k.conj().T return rho_out.real
 # Background: The generalized amplitude damping channel models the interaction of a qubit with a thermal reservoir. It is characterized by two parameters: gamma (the damping rate, 0 <= gamma <= 1) and N (the average number of thermal excitations, N >= 0). The channel is defined by four Kraus operators acting on a 2x2 Hilbert space: # A1 = sqrt(gamma * (N + 1)) * [[0, 1], [0, 0]] # A2 = sqrt(gamma * N) * [[0, 0], [1, 0]] # A3 = [[1, 0], [0, sqrt(1 - gamma) * sqrt(1 + N)]] # A4 = [[sqrt(1 - gamma) * sqrt(N), 0], [0, 1]] # These operators satisfy the completeness relation sum(Ai^dagger * Ai) = I. def generalized_amplitude_damping_channel(gamma, N): '''Generates the generalized amplitude damping channel. Inputs: gamma: float, damping parameter N: float, thermal parameter Output: kraus: list of Kraus operators as 2x2 arrays of floats, [A1, A2, A3, A4] ''' A1 = np.sqrt(gamma * (N + 1)) * np.array([[0, 1], [0, 0]]) A2 = np.sqrt(gamma * N) * np.array([[0, 0], [1, 0]]) A3 = np.array([[1, 0], [0, np.sqrt(1 - gamma) * np.sqrt(1 + N)]]) A4 = np.array([[np.sqrt(1 - gamma) * np.sqrt(N), 0], [0, 1]]) kraus = [A1, A2, A3, A4] return kraus
 # Background: The m-rail encoded state is a bipartite state in a Hilbert space of dimension m^2. Sending this state through m generalized amplitude damping channels for each party involves applying the channel operators to each of the m rails. Since the state is defined in an m-dimensional space per party, we treat each rail as a subsystem. The generalized amplitude damping channel acts on each rail (mode) independently. The total Kraus operators for the composite system are constructed by taking the tensor product of the individual channel operators across all 2*m rails. The output state is calculated by applying these Kraus operators to the initial bipartite state using the Kraus representation rho' = sum_i K_i * rho * K_i^dagger. def output_state(rails, gamma_1, N_1, gamma_2, N_2): # Initial state construction m = rails psi = np.zeros(m * m) for i in range(m): psi[i * m + i] = 1.0 / np.sqrt(m) rho = np.outer(psi, psi) # Kraus operators for a single rail def get_kraus(gamma, N): A1 = np.sqrt(gamma * (N + 1)) * np.array([[0, 1], [0, 0]]) A2 = np.sqrt(gamma * N) * np.array([[0, 0], [1, 0]]) A3 = np.array([[1, 0], [0, np.sqrt(1 - gamma) * np.sqrt(1 + N)]]) A4 = np.array([[np.sqrt(1 - gamma) * np.sqrt(N), 0], [0, 1]]) return [A1, A2, A3, A4] k1 = get_kraus(gamma_1, N_1) k2 = get_kraus(gamma_2, N_2) # The state is defined on m rails for A and m rails for B. # Here, we treat the m-rail state as a system of 2*m qubits (if m=2) or modes. # Given the problem context, we apply the channel to each rail independently. # We construct the full Kraus operators by tensoring the channel operators. # Since the state is m*m, we assume each rail is a 2-level system (qubit). full_kraus = [np.array([1.0])] for _ in range(m): new_full = [] for k in full_kraus: for op in k1: new_full.append(np.kron(k, op)) full_kraus = new_full for _ in range(m): new_full = [] for k in full_kraus: for op in k2: new_full.append(np.kron(k, op)) full_kraus = new_full rho_out = np.zeros_like(rho, dtype=complex) for k in full_kraus: rho_out += k @ rho @ k.conj().T return rho_out.real
 # Background: In m-rail encoding, the one-particle sector corresponds to states where exactly one rail is in the excited state |1> and the remaining m-1 rails are in the ground state |0>. A projector onto this sector for a single party is the sum of outer products |i><i| for i=0 to m-1, where |i> is the basis vector with a 1 at position i. For a bipartite system with m rails per party (total 2*m rails), the global projector onto the subspace where both parties are in the one-particle sector is the tensor product of the individual projectors: P_global = P_A ⊗ P_B, where P_A = sum_{i=0}^{m-1} |i><i| and P_B = sum_{j=0}^{m-1} |j><j|. def measurement(rails): '''Returns the measurement projector Input: rails: int, number of rails Output: global_proj: ( 2**rails * 2**rails, 2**rails * 2**rails ) dimensional array of floats ''' # The dimension of the Hilbert space for one party is 2^rails dim_party = 2**rails # Construct the projector for one party: P = sum_{i=0}^{rails-1} |e_i><e_i| # where |e_i> is the basis vector with 1 at index i (the i-th rail is excited) P_party = np.zeros((dim_party, dim_party)) for i in range(rails): basis_vec = np.zeros(dim_party) basis_vec[i] = 1.0 P_party += np.outer(basis_vec, basis_vec) # The global projector is the tensor product of the projectors for party A and party B global_proj = np.kron(P_party, P_party) return global_proj

 # Background: The von Neumann entropy of a quantum state represented by a density matrix rho is defined as S(rho) = -Tr(rho * log2(rho)). For a density matrix, this is calculated by finding the eigenvalues {lambda_i} of rho and computing S(rho) = -sum_{i} lambda_i * log2(lambda_i), where terms with lambda_i = 0 are treated as 0 (since lim_{x->0} x*log(x) = 0). def entropy(rho): '''Inputs: rho: 2d array of floats with equal dimensions, the density matrix of the state Output: en: quantum (von Neumann) entropy of the state rho, float ''' # Calculate eigenvalues of the density matrix eigenvalues = scipy.linalg.eigvalsh(rho) # Filter out eigenvalues that are effectively zero to avoid log(0) errors # We use a small tolerance for numerical stability nonzero_eigenvalues = eigenvalues[eigenvalues > 1e-15] # Calculate entropy using the formula S = -sum(lambda * log2(lambda)) en = -np.sum(nonzero_eigenvalues * np.log2(nonzero_eigenvalues)) return float(en)
 # Background: The coherent information of a bipartite quantum state rho_AB is defined as I(A>B) = S(rho_B) - S(rho_AB), where S(rho) = -Tr(rho * log2(rho)) is the von Neumann entropy. Here, rho_B is the reduced density matrix of system B, obtained by taking the partial trace of rho_AB over system A. The coherent information quantifies the amount of quantum information that can be reliably transmitted through a quantum channel. def coherent_inf_state(rho_AB, dimA, dimB): '''Inputs: rho_AB: 2d array of floats with equal dimensions, the state we evaluate coherent information dimA: int, dimension of system A dimB: int, dimension of system B Output co_inf: float, the coherent information of the state rho_AB ''' # Helper function for von Neumann entropy def entropy(rho): eigenvalues = scipy.linalg.eigvalsh(rho) nonzero_eigenvalues = eigenvalues[eigenvalues > 1e-15] return -np.sum(nonzero_eigenvalues * np.log2(nonzero_eigenvalues)) # Calculate S(rho_AB) s_ab = entropy(rho_AB) # Calculate rho_B = Tr_A(rho_AB) # Reshape rho_AB into a tensor (dimA, dimB, dimA, dimB) rho_tensor = rho_AB.reshape((dimA, dimB, dimA, dimB)) # Partial trace over A: sum over indices of A (i, i) rho_B = np.trace(rho_tensor, axis1=0, axis2=2) # Calculate S(rho_B) s_b = entropy(rho_B) # Coherent information I(A>B) = S(B) - S(AB) co_inf = s_b - s_ab return float(co_inf)

 # Background: The hashing protocol allows for the distillation of entanglement from a noisy bipartite state. The rate of entanglement distillation is given by the coherent information I(A>B) = S(rho_B) - S(rho_AB). In this protocol, we first pass the m-rail encoded state through generalized amplitude damping channels, then perform a post-selection measurement (projecting onto the one-particle sector for both parties). The resulting state is normalized, and its coherent information is calculated to determine the achievable entanglement rate. def rate(rails, gamma_1, N_1, gamma_2, N_2): m = rails dim_total = 2**m # 1. Construct initial state |Psi> = (1/sqrt(m)) * sum_{i=0}^{m-1} |i>_A |i>_B # Here, |i> is the basis vector with a single excitation in the i-th rail (2^m dim) psi = np.zeros(dim_total * dim_total) for i in range(m): psi[i * dim_total + i] = 1.0 / np.sqrt(m) rho = np.outer(psi, psi) # 2. Kraus operators for generalized amplitude damping def get_kraus(gamma, N): A1 = np.sqrt(gamma * (N + 1)) * np.array([[0, 1], [0, 0]]) A2 = np.sqrt(gamma * N) * np.array([[0, 0], [1, 0]]) A3 = np.array([[1, 0], [0, np.sqrt(1 - gamma) * np.sqrt(1 + N)]]) A4 = np.array([[np.sqrt(1 - gamma) * np.sqrt(N), 0], [0, 1]]) return [A1, A2, A3, A4] k1 = get_kraus(gamma_1, N_1) k2 = get_kraus(gamma_2, N_2) # 3. Apply channels to each rail # Construct full Kraus operators by tensoring individual channel operators full_kraus = [np.array([1.0])] for _ in range(m): new_full = [] for k in full_kraus: for op in k1: new_full.append(np.kron(k, op)) full_kraus = new_full for _ in range(m): new_full = [] for k in full_kraus: for op in k2: new_full.append(np.kron(k, op)) full_kraus = new_full rho_out = np.zeros_like(rho, dtype=complex) for k in full_kraus: rho_out += k @ rho @ k.conj().T # 4. Post-selection measurement P_party = np.zeros((dim_total, dim_total)) for i in range(m): basis_vec = np.zeros(dim_total) basis_vec[i] = 1.0 P_party += np.outer(basis_vec, basis_vec) global_proj = np.kron(P_party, P_party) rho_post = global_proj @ rho_out @ global_proj.T prob = np.trace(rho_post) if prob < 1e-15: return 0.0 rho_post /= prob # 5. Calculate coherent information def entropy(r): eig = scipy.linalg.eigvalsh(r) nz = eig[eig > 1e-15] return -np.sum(nz * np.log2(nz)) s_ab = entropy(rho_post) rho_tensor = rho_post.reshape((dim_total, dim_total, dim_total, dim_total)) rho_b = np.trace(rho_tensor, axis1=0, axis2=2) s_b = entropy(rho_b) return float(s_b - s_ab)
