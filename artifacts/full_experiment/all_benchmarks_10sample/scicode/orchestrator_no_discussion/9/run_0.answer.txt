import numpy as np


 # Background: The weighted Jacobi method is a stationary iterative technique for solving linear systems Ax=b. By decomposing A into D + R (where D is the diagonal component), the iteration is defined as x_{k+1} = (1-omega)x_k + omega * D^{-1}(b - Rx_k). This simplifies to the update rule x_{k+1} = x_k + omega * D^{-1}(b - Ax_k). The method converges when the spectral radius of the iteration matrix I - omega * D^{-1}A is less than 1. The convergence is monitored by the L2 norm of the increment ||x_{k+1} - x_k||_2 < eps. def WJ(A, b, eps, x_true, x0, omega): '''Solve a given linear system Ax=b with weighted Jacobi iteration method''' x = x0.astype(float) D_inv = 1.0 / np.diag(A) while True: x_new = x + omega * D_inv * (b - np.dot(A, x)) # Check convergence based on the L2 norm of the increment if np.linalg.norm(x_new - x) < eps: x = x_new break x = x_new # Calculate final residual and error residual = np.linalg.norm(np.dot(A, x) - b) error = np.linalg.norm(x - x_true) return residual, error
